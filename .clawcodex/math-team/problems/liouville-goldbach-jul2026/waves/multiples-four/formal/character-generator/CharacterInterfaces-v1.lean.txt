import Mathlib.Data.Nat.Factors
import Mathlib.Algebra.Group.Even
import Mathlib.Data.ZMod.Basic

set_option autoImplicit false

namespace ArithmeticStatement

def omega (n : ℕ) : ℕ := n.primeFactorsList.length

def lambda (n : ℕ) : ℤ := (-1 : ℤ) ^ omega n

def HasSignedRepresentation (s : ℤ) (N : ℕ) : Prop :=
  ∃ a b : ℕ,
    0 < a ∧ 0 < b ∧ N = a + b ∧
      lambda a = s ∧ lambda b = s

def HasRepresentation (N : ℕ) : Prop :=
  HasSignedRepresentation (-1 : ℤ) N

namespace FourWork

def residueLambda (p : ℕ) (z : ZMod p) : ℤ :=
  lambda z.val

def GoodMultiplier (p : ℕ) (a : ZMod p) : Prop :=
  a ≠ 0 ∧ ∀ z : ZMod p, z ≠ 0 →
    residueLambda p (a * z) = residueLambda p a * residueLambda p z

theorem residueLambda_natCast {p n : ℕ} (hnp : n < p) :
    residueLambda p (n : ZMod p) = lambda n

theorem residueLambda_sign {p : ℕ} {z : ZMod p} (hz : z ≠ 0) :
    residueLambda p z = (1 : ℤ) ∨ residueLambda p z = (-1 : ℤ)

theorem residueLambda_neg {p : ℕ} (hp : Nat.Prime p)
    (hanti : ∀ n : ℕ, 0 < n → n < p → lambda (p - n) = -lambda n)
    {z : ZMod p} (hz : z ≠ 0) :
    residueLambda p (-z) = -residueLambda p z

theorem goodMultiplier_one {p : ℕ} (hp : Nat.Prime p) :
    GoodMultiplier p (1 : ZMod p)

theorem goodMultiplier_neg {p : ℕ} (hp : Nat.Prime p)
    (hanti : ∀ n : ℕ, 0 < n → n < p → lambda (p - n) = -lambda n)
    {a : ZMod p} (ha : GoodMultiplier p a) :
    GoodMultiplier p (-a)

theorem goodMultiplier_natCast_step {p n : ℕ} (hp : Nat.Prime p)
    (hanti : ∀ j : ℕ, 0 < j → j < p → lambda (p - j) = -lambda j)
    (hn : 0 < n) (hnp : n < p)
    (hsmall : ∀ j : ℕ, 0 < j → j < n → GoodMultiplier p (j : ZMod p)) :
    GoodMultiplier p (n : ZMod p)

theorem residueLambda_mul {p : ℕ} (hp : Nat.Prime p)
    (hanti : ∀ n : ℕ, 0 < n → n < p → lambda (p - n) = -lambda n)
    {a b : ZMod p} (ha : a ≠ 0) (hb : b ≠ 0) :
    residueLambda p (a * b) = residueLambda p a * residueLambda p b

theorem lambda_eq_one_of_isSquare {p n : ℕ} (hp : Nat.Prime p)
    (hanti : ∀ j : ℕ, 0 < j → j < p → lambda (p - j) = -lambda j)
    (hn : 0 < n) (hnp : n < p) (hsquare : IsSquare (n : ZMod p)) :
    lambda n = (1 : ℤ)

theorem prime_dvd_quarter_isSquare {p r : ℕ}
    (hp : Nat.Prime p) (hLower : 7 ≤ p) (hMod : p % 4 = 3)
    (hr : Nat.Prime r) (hdiv : r ∣ (p + 1) / 4) :
    IsSquare (r : ZMod p)

theorem exists_small_prime_isSquare {p : ℕ}
    (hp : Nat.Prime p) (hLower : 7 ≤ p) (hMod : p % 4 = 3) :
    ∃ r : ℕ, Nat.Prime r ∧ r ∣ (p + 1) / 4 ∧
      r ≤ (p + 1) / 4 ∧ r < p ∧ IsSquare (r : ZMod p)

end FourWork
end ArithmeticStatement
